# --- application -----------------------------------------------------------
# 1.36+ for st.navigation. Capped because the JS is injected with
# st.components.v1.html, which is deprecated - a newer Streamlit could drop it
# and the lightbox, scroll reveals and score count-up would stop working.
streamlit>=1.36,<1.63

# --- computer vision ---------------------------------------------------------
# macOS: 0.10.14 is the last MediaPipe build for macOS and it's built against
# NumPy 1.x - with NumPy 2 it crashes ("Python quit unexpectedly"), so the
# macOS stack is pinned to match.
mediapipe==0.10.14; sys_platform == "darwin"
numpy==1.26.4; sys_platform == "darwin"
opencv-python==4.10.0.84; sys_platform == "darwin"
opencv-contrib-python==4.10.0.84; sys_platform == "darwin"
jax==0.4.30; sys_platform == "darwin"
jaxlib==0.4.30; sys_platform == "darwin"

# Linux (deployment) is pinned to the versions I tested with, otherwise a fresh
# deploy would pull MediaPipe 1.x, NumPy 2.x and OpenCV 5.x, which this code
# has never run on.
mediapipe==0.10.14; sys_platform != "darwin"
numpy==1.26.4; sys_platform != "darwin"
# Has to be opencv-contrib-python, not headless: MediaPipe depends on contrib,
# which overwrites the headless build anyway. It needs libGL, hence libgl1 in
# packages.txt.
opencv-contrib-python==4.10.0.84; sys_platform != "darwin"

# FFmpeg from pip so H.264 conversion works without a system install (a system
# ffmpeg is used first if there is one)
imageio-ffmpeg==0.6.0

# --- development -----------------------------------------------------------
# testing / linting
pytest>=8.0
ruff>=0.4
# exact version, since black's formatting changes between releases
black==26.5.1

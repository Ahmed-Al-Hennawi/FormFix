# --- application -----------------------------------------------------------
# 1.36+ for st.navigation (the / and /analyse pages).
#
# The upper bound is deliberate. The behaviour layer (utils/styling.py) installs
# scripts/main.js and scripts/analyse.js through `st.components.v1.html`, which
# Streamlit has deprecated - and `st.iframe`, its replacement, takes a URL
# rather than inline markup, so it is not a drop-in. Without a ceiling a fresh
# deploy would install a Streamlit that has dropped the API, and the reference
# lightbox, the scroll reveals and the score count-up would silently stop
# working. Raise the bound once the injection is ported.
streamlit>=1.36,<1.63

# --- computer-vision backend (Squat analyser) ------------------------------
# macOS: 0.10.14 is the LAST MediaPipe build published for macOS, and it is a
# native binary compiled against NumPy 1.x - running it with NumPy 2 crashes
# the process ("Python quit unexpectedly") during landmark detection. The
# whole macOS stack is therefore pinned to the NumPy-1 ABI it was built for.
# Linux (deployment/CI) keeps the modern stack.
mediapipe==0.10.14; sys_platform == "darwin"
numpy==1.26.4; sys_platform == "darwin"
opencv-python==4.10.0.84; sys_platform == "darwin"
opencv-contrib-python==4.10.0.84; sys_platform == "darwin"
jax==0.4.30; sys_platform == "darwin"
jaxlib==0.4.30; sys_platform == "darwin"

# Linux (deployment/CI) is pinned to the same versions the project was
# developed and tested against. Left as floors, a fresh deploy resolves to
# MediaPipe 1.x, NumPy 2.x and OpenCV 5.x - none of which this code has been
# run against, and MediaPipe 1.0 is a major release. Pinning keeps the
# deployed app the same program as the evaluated one, which matters for a
# result anyone is expected to reproduce.
mediapipe==0.10.14; sys_platform != "darwin"
numpy==1.26.4; sys_platform != "darwin"
opencv-python==4.10.0.84; sys_platform != "darwin"

# --- development -----------------------------------------------------------
# pytest, ruff and black are used for testing/linting; keep them if you want
# to run the test-suite locally:
pytest>=8.0
ruff>=0.4
# Black is pinned to an exact version, not a floor. Its formatting changes
# between releases (line-hugging rules moved in the 25.x series), so a range
# means `black --check .` passes on one machine and fails on another over code
# nobody touched. One version keeps that check meaningful.
black==26.5.1
